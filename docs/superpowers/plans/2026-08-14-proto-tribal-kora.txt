from src.kora.types import Hex, Terrain

MOVE_COST = {
    Terrain.PLAINE: 10,
    Terrain.VALLEE: 10,
    Terrain.STEPPE: 10,
    Terrain.FORET: 20,
    Terrain.COLLINE: 15,
    Terrain.MONTAGNE: 60,
    Terrain.COTE: 12,
}

BIOME_INDEX = {
    Terrain.PLAINE: 1.0,
    Terrain.VALLEE: 1.4,
    Terrain.STEPPE: 0.7,
    Terrain.FORET: 0.9,
    Terrain.COLLINE: 0.6,
    Terrain.MONTAGNE: 0.2,
    Terrain.COTE: 0.8,
    Terrain.SOMMET: 0.0,
    Terrain.EAU: 0.0,
}

NEIGHBOR_DELTAS = (
    (1, 0),
    (1, -1),
    (0, -1),
    (-1, 0),
    (-1, 1),
    (0, 1),
)


def offset_to_axial(col: int, row: int) -> Hex:
    q = col - (row - (row & 1)) // 2
    r = row
    return Hex(q, r)


def axial_to_offset(h: Hex) -> tuple[int, int]:
    col = h.q + (h.r - (h.r & 1)) // 2
    row = h.r
    return col, row


def hex_distance(a: Hex, b: Hex) -> int:
    return (abs(a.q - b.q) + abs(a.q + a.r - b.q - b.r) + abs(a.r - b.r)) // 2


def hex_neighbors(h: Hex) -> list[Hex]:
    return [Hex(h.q + dq, h.r + dr) for dq, dr in NEIGHBOR_DELTAS]


def is_passable(terrain: Terrain) -> bool:
    return terrain in MOVE_COST


def enter_cost(terrain: Terrain) -> int | None:
    return MOVE_COST.get(terrain)

import json
from pathlib import Path

from src.kora.types import Season

WIDTH = 150
HEIGHT = 100
CHAR_TO_TERRAIN = {
    "P": Terrain.PLAINE,
    "V": Terrain.VALLEE,
    "S": Terrain.STEPPE,
    "F": Terrain.FORET,
    "C": Terrain.COLLINE,
    "M": Terrain.MONTAGNE,
    "X": Terrain.SOMMET,
    "E": Terrain.EAU,
    "O": Terrain.COTE,
}
TERRAIN_TO_CHAR = {v: k for k, v in CHAR_TO_TERRAIN.items()}


class World:
    def __init__(self, terrains: list[list[Terrain]]):
        self._terrains = terrains
        self.height = len(terrains)
        self.width = len(terrains[0]) if terrains else 0
        self._exhaustion = [
            [1.0 for _ in range(self.width)] for _ in range(self.height)
        ]
        self._influence: list[list[dict[int, float]]] = [
            [{} for _ in range(self.width)] for _ in range(self.height)
        ]

    def _index(self, h: Hex) -> tuple[int, int] | None:
        col, row = axial_to_offset(h)
        if 0 <= col < self.width and 0 <= row < self.height:
            return col, row
        return None

    def in_bounds(self, h: Hex) -> bool:
        return self._index(h) is not None

    def terrain(self, h: Hex) -> Terrain:
        idx = self._index(h)
        if idx is None:
            return Terrain.EAU
        col, row = idx
        return self._terrains[row][col]

    def passable(self, h: Hex) -> bool:
        return self.in_bounds(h) and is_passable(self.terrain(h))

    def enter_cost_hex(self, h: Hex) -> int | None:
        if not self.passable(h):
            return None
        return enter_cost(self.terrain(h))

    def biome_index(self, h: Hex) -> float:
        if not self.in_bounds(h):
            return 0.0
        return BIOME_INDEX[self.terrain(h)]

    def hexes_in_radius(self, center: Hex, radius: int) -> list[Hex]:
        found: list[Hex] = []
        for dq in range(-radius, radius + 1):
            for dr in range(-radius, radius + 1):
                h = Hex(center.q + dq, center.r + dr)
                if hex_distance(center, h) <= radius and self.in_bounds(h):
                    found.append(h)
        return found

    def exhaustion(self, h: Hex) -> float:
        idx = self._index(h)
        if idx is None:
            return 1.0
        col, row = idx
        return self._exhaustion[row][col]

    def set_exhaustion(self, h: Hex, value: float) -> None:
        idx = self._index(h)
        if idx is None:
            return
        col, row = idx
        self._exhaustion[row][col] = min(1.0, max(0.5, value))

    def influence(self, h: Hex, tribe_id: int) -> float:
        idx = self._index(h)
        if idx is None:
            return 0.0
        col, row = idx
        return self._influence[row][col].get(tribe_id, 0.0)

    def add_influence(self, h: Hex, tribe_id: int, delta: float) -> None:
        idx = self._index(h)
        if idx is None:
            return
        col, row = idx
        cur = self._influence[row][col].get(tribe_id, 0.0)
        self._influence[row][col][tribe_id] = min(1.0, cur + delta)

    def scale_all_influence(self, factor: float) -> None:
        for row in self._influence:
            for cell in row:
                for tid in list(cell):
                    cell[tid] *= factor
                    if cell[tid] < 0.0001:
                        del cell[tid]


def make_filled_world(width: int, height: int, terrain: Terrain) -> World:
    grid = [[terrain for _ in range(width)] for _ in range(height)]
    return World(grid)


def paint_kora() -> World:
    grid = [[Terrain.PLAINE for _ in range(WIDTH)] for _ in range(HEIGHT)]
    import math

    for row in range(HEIGHT):
        for col in range(WIDTH):
            if col <= 3:
                grid[row][col] = Terrain.EAU
            elif col <= 8:
                grid[row][col] = Terrain.COTE
            elif row <= 4:
                grid[row][col] = Terrain.SOMMET
            elif row <= 14:
                grid[row][col] = Terrain.MONTAGNE
            elif row <= 24:
                grid[row][col] = Terrain.COLLINE
            elif row >= 85:
                grid[row][col] = Terrain.STEPPE
            elif col >= 115:
                grid[row][col] = Terrain.FORET
    for col in range(18, 125):
        row = 50 + int(5 * math.sin(col / 9.0))
        for dr in (-1, 0, 1):
            r = row + dr
            if 0 <= r < HEIGHT and grid[r][col] not in (Terrain.EAU, Terrain.SOMMET):
                grid[r][col] = Terrain.VALLEE
    return World(grid)


def save_world(world: World, path: Path) -> None:
    rows = []
    for row in range(world.height):
        chars = []
        for col in range(world.width):
            t = world._terrains[row][col]
            chars.append(TERRAIN_TO_CHAR[t])
        rows.append("".join(chars))
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps({"width": world.width, "height": world.height, "rows": rows}),
        encoding="utf-8",
    )


def load_world(path: Path) -> World:
    data = json.loads(Path(path).read_text(encoding="utf-8"))
    grid = []
    for line in data["rows"]:
        grid.append([CHAR_TO_TERRAIN[ch] for ch in line])
    return World(grid)


def season_multiplier(terrain: Terrain, season: Season) -> float:
    if season is Season.PRINTEMPS:
        base = 1.0
    elif season is Season.ETE:
        base = 1.2
    elif season is Season.AUTOMNE:
        base = 0.9
    else:
        base = 0.35
    if season is Season.HIVER and terrain in (Terrain.COLLINE, Terrain.MONTAGNE):
        return 0.20
    return base


def food_production(world: World, h: Hex, season: Season) -> float:
    if not world.in_bounds(h):
        return 0.0
    t = world.terrain(h)
    return 2.0 * BIOME_INDEX[t] * season_multiplier(t, season) * world.exhaustion(h)

---- src/kora/clock.py ----
from src.kora.types import Season


class Clock:
    def __init__(self) -> None:
        self.year = 1
        self.week = 1
        self.paused = True
        self.speed = 1
        self._finished_winter = False

    def season(self) -> Season:
        if self.week <= 13:
            return Season.PRINTEMPS
        if self.week <= 26:
            return Season.ETE
        if self.week <= 39:
            return Season.AUTOMNE
        return Season.HIVER

    def advance_week(self) -> None:
        self._finished_winter = False
        if self.week == 52:
            self.week = 1
            self.year += 1
            self._finished_winter = True
        else:
            self.week += 1

    def just_finished_winter(self) -> bool:
        return self._finished_winter

    def ticks_per_second(self) -> int:
        if self.paused:
            return 0
        return {1: 2, 2: 4, 5: 10}[self.speed]

    def set_speed(self, value: int) -> None:
        if value in (1, 2, 5):
            self.speed = value
            self.paused = False

    def toggle_pause(self) -> None:
        self.paused = not self.paused

    def weeks_until_winter(self) -> int:
        if self.week >= 40:
            return 0
        return 40 - self.week

---- src/kora/path.py ----
import heapq

from src.kora.types import Hex
from src.kora.world import hex_distance, hex_neighbors, World

MOVE_POINTS_PER_WEEK = 60


def astar(world: World, start: Hex, goal: Hex) -> list[Hex] | None:
    if start == goal:
        return []
    if not world.passable(start) or not world.passable(goal):
        return None
    open_heap: list[tuple[int, int, Hex]] = []
    seq = 0
    heapq.heappush(open_heap, (0, seq, start))
    came: dict[Hex, Hex] = {}
    g: dict[Hex, int] = {start: 0}
    closed: set[Hex] = set()
    while open_heap:
        _f, _, current = heapq.heappop(open_heap)
        if current in closed:
            continue
        if current == goal:
            path: list[Hex] = []
            node = current
            while node != start:
                path.append(node)
                node = came[node]
            path.reverse()
            return path
        closed.add(current)
        for nb in hex_neighbors(current):
            cost = world.enter_cost_hex(nb)
            if cost is None:
                continue
            tentative = g[current] + cost
            if tentative < g.get(nb, 10**9):
                came[nb] = current
                g[nb] = tentative
                seq += 1
                f = tentative + hex_distance(nb, goal) * 10
                heapq.heappush(open_heap, (f, seq, nb))
    return None


def travel_weeks(world: World, start: Hex, path: list[Hex]) -> int:
    if not path:
        return 0
    weeks = 0
    points = 0
    remaining = list(path)
    while remaining:
        if points <= 0:
            weeks += 1
            points += MOVE_POINTS_PER_WEEK
        nxt = remaining[0]
        cost = world.enter_cost_hex(nxt)
        if cost is None:
            return weeks
        if cost > points:
            points = 0
            continue
        points -= cost
        remaining.pop(0)
        _ = start
    return weeks
---- fin ----

AI_STEPPE_ID = 2
AI_FOREST_ID = 3
AI_COAST_ID = 4
FORAGE_RADIUS = 2
STOCK_MAX_FACTOR = 4


@dataclass
class GameState:
    world: World
    clock: Clock
    tribes: dict[int, Tribe]
    bands: dict[int, Band]
    tick_count: int = 0
    rng: random.Random = field(default_factory=lambda: random.Random(1))
    last_error: str | None = None
    player_dead: bool = False
    last_pressure: dict[Hex, float] = field(default_factory=dict)


def stock_max(band: Band) -> float:
    return STOCK_MAX_FACTOR * band.population


def forage_hexes(state: GameState, band: Band) -> list[Hex]:
    return state.world.hexes_in_radius(band.position, FORAGE_RADIUS)


def collect_food(state: GameState) -> None:
    season = state.clock.season()
    claimants: dict[Hex, list[Band]] = {}
    for band in state.bands.values():
        if band.population <= 0:
            continue
        for h in forage_hexes(state, band):
            claimants.setdefault(h, []).append(band)
    gained: dict[int, float] = {bid: 0.0 for bid in state.bands}
    pressure: dict[Hex, float] = {}
    for h, bands in claimants.items():
        prod = food_production(state.world, h, season)
        total_pop = sum(b.population for b in bands)
        pressure[h] = sum(b.population / 19.0 for b in bands)
        if total_pop <= 0 or prod <= 0:
            continue
        for b in bands:
            gained[b.id] += prod * (b.population / total_pop)
    for band in state.bands.values():
        take = gained.get(band.id, 0.0)
        need = band.population * 1.0
        available = band.stock + take
        if available >= need:
            band.stock = min(stock_max(band), available - need)
        else:
            deficit = need - available
            band.stock = 0.0
            loss = max(1, math.floor(deficit)) if deficit > 0 else 0
            band.population = max(0, band.population - loss)
            band.famine_in_period = True
    state.last_pressure = pressure


def update_exhaustion(state: GameState) -> None:
    season = state.clock.season()
    for row in range(state.world.height):
        for col in range(state.world.width):
            h = offset_to_axial(col, row)
            prod = food_production(state.world, h, season)
            p = state.last_pressure.get(h, 0.0)
            if p > 2.0 * prod and prod > 0:
                state.world.set_exhaustion(h, 0.5)
            else:
                state.world.set_exhaustion(
                    h, min(1.0, state.world.exhaustion(h) + 0.1)
                )

from src.kora.path import MOVE_POINTS_PER_WEEK, astar
from src.kora.types import Order, OrderKind


def set_goto(state: GameState, band_id: int, goal: Hex) -> None:
    band = state.bands[band_id]
    if not state.world.passable(goal):
        return
    path = astar(state.world, band.position, goal)
    if path is None:
        return
    band.order = Order(kind=OrderKind.GOTO, target_hex=goal)
    band.path = path


def set_march_to_band(state: GameState, band_id: int, target_band_id: int) -> None:
    if target_band_id not in state.bands or band_id == target_band_id:
        return
    band = state.bands[band_id]
    target = state.bands[target_band_id]
    path = astar(state.world, band.position, target.position)
    if path is None:
        return
    band.order = Order(
        kind=OrderKind.MARCH_TO_BAND, target_band_id=target_band_id
    )
    band.path = path


def apply_movement(state: GameState) -> None:
    for band in state.bands.values():
        if band.population <= 0:
            continue
        if band.order.kind is OrderKind.MARCH_TO_BAND:
            tid = band.order.target_band_id
            if tid not in state.bands:
                band.order = stay_order()
                band.path = []
                continue
            target = state.bands[tid]
            path = astar(state.world, band.position, target.position)
            band.path = [] if path is None else path
        points = MOVE_POINTS_PER_WEEK
        while band.path and points > 0:
            nxt = band.path[0]
            cost = state.world.enter_cost_hex(nxt)
            if cost is None or cost > points:
                break
            points -= cost
            band.position = nxt
            band.path.pop(0)
        if not band.path and band.order.kind is OrderKind.GOTO:
            band.order = stay_order()

def test_vision_radius_hides_far_enemy():
    world = make_filled_world(40, 40, Terrain.PLAINE)
    p = offset_to_axial(10, 10)
    far = offset_to_axial(10, 30)
    near = offset_to_axial(12, 12)
    pb = Band(id=1, tribe_id=1, position=p, population=10, stock=40)
    enemy_far = Band(id=2, tribe_id=2, position=far, population=10, stock=40)
    enemy_near = Band(id=3, tribe_id=2, position=near, population=10, stock=40)
    st = GameState(
        world=world,
        clock=Clock(),
        tribes={
            1: Tribe(1, "p", 20, True),
            2: Tribe(2, "e", 20, False),
        },
        bands={1: pb, 2: enemy_far, 3: enemy_near},
    )
    vis = recompute_vision(st)
    assert hex_distance(p, far) >= 20
    assert far not in vis.visible
    assert hex_distance(p, near) <= 8
    assert near in vis.visible

Le calendrier avance EN DEBUT de tick : la premiere semaine jouee
devient semaine 2 si on avance d'abord. Pour rester fidele a "an 1
semaine 1 visible au lancement", new_game laisse week=1, et tick
avance ensuite. Le premier tick fait passer a semaine 2 apres avoir
simule... Spec : "Avancer le calendrier" en premier. Acceptable :
l'affichage montre l'etat AVANT tick ; le joueur voit semaine 1
en pause ; au unpause le premier tick passe a 2. Documente ici.

Alternative plus naturelle : avancer le calendrier EN FIN de tick.
La spec dit le contraire. Suivre la spec (avance en premier).
new_game : week=1 pause. Premier tick -> week=2. OK.
